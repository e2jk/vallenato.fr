# Hash-pinned pip upgrade, installed before anything else in a fresh venv or
# the production image.
#
# `python -m venv` bundles whatever pip version ships with that Python
# build's ensurepip - not necessarily current, and potentially vulnerable
# (this pin exists because pip 25.0.1, what actually got bundled in CI,
# carried 7 known CVEs needing 26.2+ to clear). Hash-pinned rather than a
# bare `pip install --upgrade pip` so this satisfies the same
# Pinned-Dependencies expectation as bin/requirements.txt, and so Renovate
# (which manages bin/requirements*.txt) keeps it current automatically
# instead of a version silently going stale here.
pip==26.2.2 \
    --hash=sha256:3a8e5c800492787f33a4114ef54444546c340df428974ebba17e0fd47e11af2f \
    --hash=sha256:fd7f4dfa8190e014013b9f08f3e973f4e4c26ffd179a204895533a7254813a70
